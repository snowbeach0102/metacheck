아래 JSON 하나만 출력한다. 필드는 이 순서대로 쓰고, 주석(//)은 출력하지 않는다. [예시]와 같은 모양이다.
{
  "questions": [
    {
      "c": "개념 id",
      "lv": 1 | 2 | 3,
      "concepts": ["문제에 쓴 개념 이름", ...],   // 기초 1개 · 실전 1~2개 · 심화 2개 이상
      "steps": ["풀이 추론 단계", ...],            // 기초 [] · 실전 1~2개 · 심화 3개 이상. 먼저 쓴다
      "q": "지문",
      "o": ["보기", "보기", "보기", "보기"],
      "a": 정답 위치(0~3),
      "shuffle": true | false,                     // 숫자 · 순서 · 조합 보기면 false
      "distractors": { "오답 위치": "노린 오개념 · 실수 한 줄", ... },   // 정답을 뺀 3개
      "x": "해설",
      "ev": "정답 근거 — 자료의 문장 또는 핵심 원리"
    }
  ],
  "skipped": [ { "c": "개념 id", "reason": "만들 수 없는 이유" } ]   // 없으면 []
}
